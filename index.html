/* LOGIC HUẤN LUYỆN TENSORFLOW KHÔNG LÀM TREO TRÌNH DUYỆT */ async function trainAndPredict() { const btn = document.getElementById('btn-train'); const statusEl = document.getElementById('training-status'); const progressEl = document.getElementById('training-progress'); const gameType = document.getElementById('game-type').value; const algoType = document.getElementById('algo-type').value; btn.disabled = true; btn.classList.add('opacity-50', 'cursor-not-allowed'); let maxNum = 45; if (gameType === 'power655') maxNum = 55; if (gameType === 'lotto535') maxNum = 35; statusEl.innerText = "Chuẩn hóa dữ liệu..."; progressEl.style.width = '10%'; // Nhường luồng cho UI vẽ lại thanh progress bar await tf.nextFrame(); const inputs = []; const labels = []; if (['mega645', 'power655', 'lotto535'].includes(gameType)) { for (let i = 0; i < rawDraws.length - 1; i++) { const currentDraw = rawDraws[i].numbers; const nextDraw = rawDraws[i+1].numbers; const inVec = new Array(maxNum).fill(0); currentDraw.forEach(n => { if (n <= maxNum) inVec[n-1] = 1; }); const outVec = new Array(maxNum).fill(0); nextDraw.forEach(n => { if (n <= maxNum) outVec[n-1] = 1; }); inputs.push(inVec); labels.push(outVec); } } else { for (let i = 0; i < rawDraws.length - 1; i++) { const inVec = new Array(10).fill(0); rawDraws[i].numbers.forEach(s => s.split('').forEach(d => inVec[parseInt(d)]++)); const outVec = new Array(10).fill(0); rawDraws[i+1].numbers.forEach(s => s.split('').forEach(d => outVec[parseInt(d)]++)); inputs.push(inVec); labels.push(outVec); } } const featureSize = inputs[0].length; const model = tf.sequential(); // Phân nhánh kiến trúc AI dựa trên Lựa chọn Thuật toán if (algoType === 'tf-lstm') { // Pseudo-LSTM / Reshaped Pattern Neural Net model.add(tf.layers.dense({ units: 128, activation: 'relu', inputShape: [featureSize] })); model.add(tf.layers.repeatVector({ n: 2 })); model.add(tf.layers.lstm({ units: 64, returnSequences: false })); model.add(tf.layers.dense({ units: featureSize, activation: 'sigmoid' })); } else { // Standard Dense Neural Network model.add(tf.layers.dense({ units: 64, activation: 'relu', inputShape: [featureSize] })); model.add(tf.layers.dropout({ rate: 0.2 })); model.add(tf.layers.dense({ units: 32, activation: 'relu' })); model.add(tf.layers.dense({ units: featureSize, activation: 'sigmoid' })); } model.compile({ optimizer: tf.train.adam(0.01), loss: 'binaryCrossentropy' }); statusEl.innerText = "Đang huấn luyện AI..."; progressEl.style.width = '30%'; await tf.nextFrame(); // Chuyển dữ liệu sang Tensor const xs = tf.tensor2d(inputs); const ys = tf.tensor2d(labels); const epochs = 20; // Huấn luyện không gây treo UI nhờ tf.nextFrame() await model.fit(xs, ys, { epochs: epochs, batchSize: 16, callbacks: { onEpochEnd: async (epoch, logs) => { const pct = Math.floor(30 + ((epoch + 1) / epochs) * 60); progressEl.style.width = `${pct}%`; statusEl.innerText = `Epoch ${epoch + 1}/${epochs}`; // GIẢI PHÁP THEN CHỐC: Nhường luồng giao diện sau mỗi epoch await tf.nextFrame(); } } }); statusEl.innerText = "Dự đoán bộ số..."; progressEl.style.width = '95%'; await tf.nextFrame(); // Tiến hành dự đoán và dọn dẹp bộ nhớ Tensor const lastDrawVec = inputs[0]; const predictionData = tf.tidy(() => { const inputTensor = tf.tensor2d([lastDrawVec]); const outputTensor = model.predict(inputTensor); return outputTensor.dataSync(); }); // Giải phóng bộ nhớ Tensor & Model xs.dispose(); ys.dispose(); model.dispose(); progressEl.style.width = '100%'; statusEl.innerText = "Hoàn tất!"; // Hiển thị kết quả renderAIPredictions(predictionData, gameType, currentTicketCount, maxNum); updateJackpotMeter(); triggerFireworks(); btn.disabled = false; btn.classList.remove('opacity-50', 'cursor-not-allowed'); }
